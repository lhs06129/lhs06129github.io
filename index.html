// 기존 사운드 관련 변수 및 함수 교체 let audioCtx = null; let currentGainNode = null; let currentOscillators = []; function ensureAudioContext() { if (!audioCtx) { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } if (audioCtx.state === 'suspended') { audioCtx.resume(); } } const dtmfFrequencies = { '1': [697, 1209], '2': [697, 1336], '3': [697, 1477], '4': [770, 1209], '5': [770, 1336], '6': [770, 1477], '7': [852, 1209], '8': [852, 1336], '9': [852, 1477], '*': [941, 1209], '0': [941, 1336], '#': [941, 1477] }; function playSound(keyVal) { ensureAudioContext(); stopSound(); // 이전 소리가 남아있다면 정리가 완전히 끝난 후 재생 const freqs = dtmfFrequencies[keyVal]; if (!freqs || !audioCtx) return; const now = audioCtx.currentTime; // GainNode(부드러운 볼륨 조절기) 생성 currentGainNode = audioCtx.createGain(); // 시작 시 0에서 0.025초 동안 0.08 볼륨으로 자연스럽게 페이드 인 currentGainNode.gain.setValueAtTime(0, now); currentGainNode.gain.linearRampToValueAtTime(0.08, now + 0.025); currentGainNode.connect(audioCtx.destination); // 오실레이터 생성 currentOscillators = freqs.map(freq => { const osc = audioCtx.createOscillator(); osc.type = 'sine'; osc.frequency.setValueAtTime(freq, now); osc.connect(currentGainNode); osc.start(now); return osc; }); } function stopSound() { if (currentGainNode && audioCtx) { const now = audioCtx.currentTime; const fadeTime = 0.03; // 0.03초 동안 부드럽게 감쇄(Fade Out) // 쨍하는 소리가 나지 않도록 볼륨을 0으로 부드럽게 깎음 currentGainNode.gain.cancelScheduledValues(now); currentGainNode.gain.setValueAtTime(currentGainNode.gain.value, now); currentGainNode.gain.linearRampToValueAtTime(0, now + fadeTime); const oscsToStop = [...currentOscillators]; setTimeout(() => { oscsToStop.forEach(osc => { try { osc.stop(); } catch(e) {} }); }, fadeTime * 1000 + 10); currentOscillators = []; currentGainNode = null; } }
