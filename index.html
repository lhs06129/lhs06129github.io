/* 1. iOS DTMF 표준 키패드 오디오 파일 매핑 (Base64/외부 리소스 연동) */ // 무음 파일로 오디오 채널 프리워밍(Pre-warming) const silentAudio = new Audio("data:audio/wav;base64,UklGRigAAABXQVZFZm10IBAAAAABAAEARKwAAIhYAQACABAAZGF0YQAAAAA="); // 모든 사운드 미리 로드 (Preload) Object.values(dtmfSounds).forEach(audio => { audio.preload = "auto"; }); let isAudioWarmedUp = false; /* 2. 첫 터치 시 인지하지 못하는 무음 재생으로 오디오 채널 즉시 개방 */ function warmUpAudio() { if (isAudioWarmedUp) return; silentAudio.play().then(() => { isAudioWarmedUp = true; }).catch(() => {}); } // 화면 진입 후 어디든 첫 터치/클릭하는 순간 무음 재생 실행 ['touchstart', 'touchend', 'mousedown', 'pointerdown'].forEach(evt => { window.addEventListener(evt, warmUpAudio, { once: true, capture: true }); }); /* 3. 소리 재생 및 정지 로직 */ let currentAudio = null; function playSound(keyVal) { warmUpAudio(); // 첫 타 보장 if (currentAudio) { currentAudio.pause(); currentAudio.currentTime = 0; } const sound = dtmfSounds[keyVal]; if (sound) { currentAudio = sound; currentAudio.currentTime = 0; currentAudio.volume = 0.5; currentAudio.play().catch(() => {}); } } function stopSound() { if (currentAudio) { // 손을 뗐을 때 즉시 멈춤 currentAudio.pause(); currentAudio.currentTime = 0; currentAudio = null; } } /* 4. 터치 이벤트 연동 */ keys.forEach(key => { const keyVal = key.getAttribute("data-key"); const pressStart = (e) => { key.classList.add("active"); playSound(keyVal); }; const pressEnd = (e) => { if (key.classList.contains("active")) { key.classList.remove("active"); stopSound(); handleKeyPress(); } }; const pressCancel = () => { key.classList.remove("active"); stopSound(); }; key.addEventListener("touchstart", (e) => { e.preventDefault(); pressStart(e); }, { passive: false }); key.addEventListener("touchend", (e) => { e.preventDefault(); pressEnd(e); }, { passive: false }); key.addEventListener("touchcancel", pressCancel); key.addEventListener("mousedown", pressStart); key.addEventListener("mouseup", pressEnd); key.addEventListener("mouseleave", pressCancel); }); function handleKeyPress() { if (index < targetNumber.length) { index++; let value = targetNumber.substring(0, index); if (value.length > 3 && value.length <= 7) { value = value.substring(0, 3) + "-" + value.substring(3); } else if (value.length > 7) { value = value.substring(0, 3) + "-" + value.substring(3, 7) + "-" + value.substring(7); } display.textContent = value; } } contactsTab.addEventListener("click", () => { const input = prompt("마술용 전화번호를 입력하세요 (숫자만):", targetNumber); if (input !== null && input.trim() !== "") { targetNumber = input.replace(/[^0-9]/g, ""); index = 0; display.textContent = ""; alert("전화번호가 " + targetNumber + " (으)로 변경되었습니다."); } }); call.addEventListener("touchstart", (e) => { e.preventDefault(); call.classList.add("active"); }); call.addEventListener("touchend", (e) => { e.preventDefault(); call.classList.remove("active"); if (index > 0) { window.location.href = "tel:" + targetNumber; } }); call.addEventListener("mousedown", () => call.classList.add("active")); call.addEventListener("mouseup", () => { call.classList.remove("active"); if (index > 0) { window.location.href = "tel:" + targetNumber; } });
